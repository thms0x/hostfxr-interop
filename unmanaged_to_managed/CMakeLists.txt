cmake_minimum_required(VERSION 3.22)
project(unmanaged_to_managed VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/DotNet.cmake")
find_package(Threads REQUIRED)

add_library(interop STATIC src/hosting.cpp src/interop.cpp)
add_library(interop::interop ALIAS interop)
set_target_properties(interop PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(interop PUBLIC cxx_std_17)
target_include_directories(interop PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_link_libraries(interop PRIVATE Threads::Threads)
if(INTEROP_TARGET_FRAMEWORK STREQUAL "net48")
    target_compile_definitions(interop PRIVATE INTEROP_NETFRAMEWORK)
    target_link_libraries(interop PRIVATE mscoree)
else()
    target_compile_definitions(interop PRIVATE NETHOST_USE_AS_STATIC)
    target_link_libraries(interop PRIVATE interop_nethost ${CMAKE_DL_LIBS})
    if(WIN32)
        target_link_libraries(interop PRIVATE advapi32)
    endif()
endif()
if(MSVC)
    target_compile_options(interop PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(interop PRIVATE -Wall -Wextra -Wpedantic)
endif()
interop_managed(interop_bridge "${CMAKE_CURRENT_SOURCE_DIR}/managed/Interop.Managed" Interop.Managed)
add_dependencies(interop interop_bridge)

add_executable(interop_example examples/main.cpp)
target_link_libraries(interop_example PRIVATE interop::interop)
add_dependencies(interop_example interop_bridge)
